cmake_minimum_required(VERSION 3.22.1)
project(BBYQ_XI_Lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(BBYQ_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../" ABSOLUTE)
set(REVERSE "${BBYQ_ROOT}/deps/reverse-pc")
set(MUNDO "${BBYQ_ROOT}/deps/mundo-x")
set(PATCH "${BBYQ_ROOT}/build/ether-compat")
if(NOT EXISTS "${REVERSE}/src/rvpc2.hpp" OR NOT EXISTS "${MUNDO}/native/include/mundo_x/quantum_runtime.hpp" OR NOT EXISTS "${PATCH}/mundo_x/quantum_runtime.hpp")
    message(FATAL_ERROR "Missing original pinned XI engine and Mundo X Ether overlay")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${REVERSE}" OUTPUT_VARIABLE REVERSE_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${MUNDO}" OUTPUT_VARIABLE MUNDO_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT REVERSE_SHA STREQUAL "a1b935ae0adf38615996bac92727e95289ad7cc1" OR NOT MUNDO_SHA STREQUAL "e2778d962af22f6f1a0794058dd4685fc2573966")
    message(FATAL_ERROR "Unverified REVERSE-PC/Mundo X checkout")
endif()
add_library(reverse_os SHARED "${REVERSE}/reverse-os-android/app/src/main/cpp/reverse_os_jni.cpp")
target_include_directories(reverse_os PRIVATE
    "${PATCH}" "${MUNDO}/native/include" "${REVERSE}/src"
    "${REVERSE}/reverse-os-android/app/src/main/cpp")
target_compile_options(reverse_os PRIVATE -O2 -fexceptions -frtti -Wall -Wextra -Wpedantic)
target_compile_definitions(reverse_os PRIVATE REVERSE_PC_HAVE_BOUNDED_PRODUCT_IMPORT=1)
find_library(LOG_LIBRARY log)
target_link_libraries(reverse_os PRIVATE ${LOG_LIBRARY})
